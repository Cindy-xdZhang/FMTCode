cmake_minimum_required(VERSION 3.18)
project(task6_ggt17_observer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(OpenMP REQUIRED)
add_library(reference SHARED reference.cpp)
target_include_directories(reference PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/third_party/eigen")
target_link_libraries(reference PRIVATE OpenMP::OpenMP_CXX)
target_compile_definitions(reference PRIVATE EIGEN_NO_DEBUG NOMINMAX)
if(MSVC)
  target_compile_options(reference PRIVATE /O2 /bigobj)
else()
  target_compile_options(reference PRIVATE -O2)
endif()
